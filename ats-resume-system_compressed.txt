from pydantic import BaseModel, Field
from typing import List, Optional

class PersonalInfo(BaseModel):
    fullName: str
    email: str
    phone: Optional[str] = None
    location: Optional[str] = None
    linkedinUrl: Optional[str] = None
    githubUrl: Optional[str] = None
    portfolioUrl: Optional[str] = None
    summary: Optional[str] = None

class Skills(BaseModel):
    technical: List[str] = []
    tools: List[str] = []
    softSkills: List[str] = []

class Project(BaseModel):
    title: str
    techStack: List[str] = []
    liveLink: Optional[str] = None
    repoLink: Optional[str] = None
    bullets: List[str] = []

class OptimizedProject(BaseModel):
    title: str
    techStack: List[str] = []
    liveLink: Optional[str] = None
    repoLink: Optional[str] = None
    optimizedBullets: List[str] = []

class Education(BaseModel):
    institution: str
    degree: str
    fieldOfStudy: Optional[str] = None
    gradYear: Optional[str] = None
    score: Optional[str] = None

class ExperienceItem(BaseModel):
    title: str
    organization: Optional[str] = None
    date: Optional[str] = None
    description: Optional[str] = None

class Certification(BaseModel):
    name: str
    issuingOrganization: Optional[str] = None
    issueDate: Optional[str] = None
    credentialUrl: Optional[str] = None

class Achievement(BaseModel):
    title: str
    description: Optional[str] = None
    date: Optional[str] = None

class ProfileSchemaObject(BaseModel):
    personalInfo: PersonalInfo
    skills: Skills
    experience: List[ExperienceItem] = []
    projects: List[Project] = []
    education: List[Education] = []
    certifications: List[Certification] = []
    achievements: List[Achievement] = []

class TailoredSummaryOutput(BaseModel):
    tailoredSummary: str

class PrioritizedSkillsOutput(BaseModel):
    technical: List[str]
    tools: List[str]
    softSkills: List[str]

class OptimizedProjectsOutput(BaseModel):
    optimizedProjects: List[OptimizedProject]

class ScoreBreakdown(BaseModel):
    keywordMatchScore: float
    relevanceScore: float
    impactMetricScore: float
    formattingStructureScore: float

class Scoring(BaseModel):
    overallMatchScore: float = Field(..., ge=0, le=100)
    breakdown: ScoreBreakdown

class ProItem(BaseModel):
    point: str
    category: str

class ConItem(BaseModel):
    issue: str
    severity: str 
    suggestion: str

class AuditAndScoringOutput(BaseModel):
    targetJobTitle: str = Field(default="Full Stack - Web Developer")
    scoring: Scoring
    pros: List[ProItem]
    cons: List[ConItem]
    missingKeywords: List[str]

class AuditFeedback(BaseModel):
    pros: List[ProItem]
    cons: List[ConItem]
    missingKeywords: List[str]

class TailoredResumeContent(BaseModel):
    personalInfo: PersonalInfo
    tailoredSummary: str
    prioritizedSkills: Skills
    experience: List[ExperienceItem] = []
    optimizedProjects: List[OptimizedProject]
    education: List[Education] = []
    certifications: List[Certification] = []
    achievements: List[Achievement] = []

class AnalysisResponseSchema(BaseModel):
    targetJobTitle: str = Field(default="Full Stack - Web Developer")
    scoring: Scoring
    audit: AuditFeedback
    tailoredContent: TailoredResumeContent

import os
import re
import json
import asyncio
import traceback
from fastapi import FastAPI, HTTPException
from groq import AsyncGroq
from dotenv import load_dotenv
from schemas import (
    ProfileSchemaObject,
    AnalysisResponseSchema,
    TailoredSummaryOutput,
    PrioritizedSkillsOutput,
    OptimizedProjectsOutput,
    AuditAndScoringOutput,
    AuditFeedback,
    TailoredResumeContent
)
from pydantic_core import to_jsonable_python

load_dotenv()
api_key = os.getenv("GROQ_API_KEY")

if not api_key:
    raise RuntimeError("GROQ_API_KEY is missing from environment variables.")

client = AsyncGroq(api_key=api_key)
app = FastAPI(title="ATS AI Microservice", version="1.0")

MODELS = ["openai/gpt-oss-120b", "openai/gpt-oss-20b"]

async def _groq_structured(system_prompt: str, user_prompt: str, schema_class, temperature: float = 0.1):
    clean_schema = to_jsonable_python(schema_class.model_json_schema())
    schema_str = json.dumps(clean_schema)
    full_system = f"{system_prompt}\n\nSTRICT REQUIREMENT: Output MUST be valid JSON matching this schema:\n{schema_str}"
    last_err = None
    for model_name in MODELS:
        try:
            print(f"[*] Calling {model_name}...")
            response = await client.chat.completions.create(
                model=model_name,
                messages=[
                    {"role": "system", "content": full_system},
                    {"role": "user", "content": str(user_prompt)}
                ],
                response_format={"type": "json_object"},
                temperature=temperature
            )
            raw_content = response.choices[0].message.content
            return json.loads(raw_content)
        except Exception as e:
            last_err = e
            print(f"[!] Warning on {model_name}: {e}")
            await asyncio.sleep(1.0)
            continue
            
    raise RuntimeError(f"All Groq models failed. Last error: {last_err}")

@app.get("/")

@app.post("/ai/parse-linkedin")
async def parse_linkedin(payload: dict):
    raw_text = payload.get("rawText")
    if not raw_text:
        raise HTTPException(status_code=400, detail="rawText is required.")
    
    system_prompt = """You are an expert HR data parser.
STRICT ACCURACY RULES:
1. Extract ALL actual work experience entries into the 'experience' array.
2. If phone, linkedin, or github are NOT in the text, you MUST output null. NEVER invent mock contact details.
3. If candidate has no experience, output 'experience': []."""
    user_prompt = f"Extract all details into JSON from this raw LinkedIn text:\n{raw_text}"
    
    return await _groq_structured(system_prompt, user_prompt, ProfileSchemaObject, temperature=0.0)

async def generate_summary(candidate_summary: str, skills: dict, jd: str):
    system_prompt = """You are an elite ATS Resume Optimization Specialist and Technical Recruiter.
Your objective is to craft a high-impact, ATS-optimized Professional Summary.

Strict Compliance Rules:
1. FACT PRESERVATION: NEVER invent or inflate years of experience, credentials, or unverified technical tools.
2. NO EMPTY BUZZWORDS: Avoid generic adjectives like 'hard-working', 'passionate', or 'enthusiastic'.
3. LENGTH & STRUCTURE: Exactly 2 to 3 concise sentences highlighting core verified stack and engineering capability.
4. FORBIDDEN: NEVER mention salary, pay scale, remuneration, or 'UGC norms' anywhere in the output."""
    user_prompt = f"Candidate Current Summary: {candidate_summary}\nCandidate Skills: {json.dumps(skills)}\nTarget JD:\n{jd}"
    return await _groq_structured(system_prompt, user_prompt, TailoredSummaryOutput, temperature=0.2)

async def generate_prioritized_skills(candidate_skills: dict, jd: str):
    system_prompt = """You are an elite ATS Technical Resume Categorizer and Keyword Strategist.
Your task is to categorize verified skills into 3 STRICT technical buckets:
1. 'technical': Programming Languages, Runtimes & Core Web Frameworks.
2. 'tools': Developer Tools, Databases, Platforms & Infrastructure.
3. 'softSkills': Core Technical Concepts & Methodologies (DSA, OOP, System Design).
NEGATIVE CONSTRAINT: Drop all personal hobbies, photography, sports, or unrelated trades completely."""
    user_prompt = f"Candidate Skills: {json.dumps(candidate_skills)}\nTarget JD:\n{jd}"
    return await _groq_structured(system_prompt, user_prompt, PrioritizedSkillsOutput, temperature=0.1)

async def generate_optimized_projects(candidate_projects: list, jd: str):
    system_prompt = """You are an elite Technical Resume Architect.
Optimize project bullet points to maximize ATS ranking and technical authority:
1. FACT PRESERVATION: NEVER invent fake numerical metrics (e.g. do not invent 'scaled to 100k users').
2. POWER VERB DISCIPLINE: Every bullet must start with a strong action verb (Architected, Engineered, Implemented, Streamlined).
3. GOOGLE X-Y-Z FORMULA: 'Accomplished [X] using [Y], resulting in [Z]'.
4. Exactly 2 to 3 focused bullet points per project. NEVER invent new project titles or fake links."""
    user_prompt = f"Candidate Projects: {json.dumps(candidate_projects)}\nTarget JD:\n{jd}"
    return await _groq_structured(system_prompt, user_prompt, OptimizedProjectsOutput, temperature=0.2)

async def generate_audit_and_score(profile_data: dict, jd: str):
    system_prompt = """You are an enterprise-grade Applicant Tracking System (ATS) Parser and Principal Technical Recruiter.
Conduct a mathematically rigorous gap analysis:

1. DYNAMIC JOB TITLE EXTRACTION: Extract the exact target job title/role targeted in the JD (e.g., 'Frontend Developer', 'Assistant Professor - Computer Science', 'DevOps Engineer').
   - MUST be 2 to 5 words in Title Case.
   - FORBIDDEN: NEVER output 'About the job', 'Job Description', 'UGC Norms', or company names.
2. EXPERIENCE SURPLUS RULE: If the JD asks for X years (e.g. 3 years) and the candidate has MORE experience, that is a PRO (strength), NEVER a con or gap.
3. DURATION IS NOT A KEYWORD: NEVER output numbers or durations (e.g., '3 years', '5+ years') in missingKeywords.
4. missingKeywords must ONLY contain concrete missing technical skills, tools, or frameworks.
5. FORBIDDEN: Do NOT mention salary, pay scale, or UGC norms."""
    user_prompt = f"Candidate Profile:\n{json.dumps(profile_data)}\n\nTarget Job Description:\n{jd}"
    return await _groq_structured(system_prompt, user_prompt, AuditAndScoringOutput, temperature=0.1)

@app.post("/ai/analyze-and-tailor", response_model=AnalysisResponseSchema)
async def analyse_and_tailor(payload: dict):
    profile_data = payload.get("profileData")
    job_description = payload.get("jobDescription")

    if not profile_data or not job_description:
        raise HTTPException(status_code=400, detail="profileData and jobDescription are required.")

    personal_info = profile_data.get("personalInfo", {})
    existing_skills = profile_data.get("skills", {})
    existing_projects = profile_data.get("projects", [])
    existing_experience = profile_data.get("experience", [])

    try:
        summary_res = await generate_summary(personal_info.get("summary", ""), existing_skills, job_description)
        skills_res = await generate_prioritized_skills(existing_skills, job_description)
        projects_res = await generate_optimized_projects(existing_projects, job_description)
        audit_res = await generate_audit_and_score(profile_data, job_description)

        # Python-safe dynamic title cleaning (replaces JS regex bug)
        raw_title = audit_res.get("targetJobTitle") or ""
        extracted_title = str(raw_title).strip()
        if not extracted_title or re.search(r"about the job|job description|ugc norms", extracted_title, re.IGNORECASE):
            extracted_title = "Full Stack - Web Developer"

        return AnalysisResponseSchema(
            targetJobTitle=extracted_title,
            scoring=audit_res["scoring"],
            audit=AuditFeedback(
                pros=audit_res["pros"],
                cons=audit_res["cons"],
                missingKeywords=audit_res["missingKeywords"]
            ),
            tailoredContent=TailoredResumeContent(
                personalInfo=personal_info,
                tailoredSummary=summary_res["tailoredSummary"],
                prioritizedSkills=skills_res,
                experience=existing_experience,
                optimizedProjects=projects_res["optimizedProjects"],
                education=profile_data.get("education", []),
                certifications=profile_data.get("certifications", []),
                achievements=profile_data.get("achievements", [])
            )
        )
    except Exception as e:
        traceback.print_exc()
        raise HTTPException(status_code=500, detail=f"Pipeline error: {str(e)}")

import React, { useState, useEffect, useRef } from 'react';

const formatJoin = (val) => (Array.isArray(val) ? val.filter(Boolean).join(', ') : val || '');
const toArray = (val) => (Array.isArray(val) ? val.filter(Boolean) : val ? [val] : []);
const toText = (val) => (typeof val === 'string' ? val.trim() : val != null ? String(val) : '');
const cleanUrl = (url) => (typeof url === 'string' ? url.replace(/^https?:\/\/(www\.)?/, '').replace(/\/$/, '') : '');

const resolveTarget = (con) => {
  if (con.targetField) return con.targetField;
  const issue = (con.issue || '').toLowerCase();
  
  if (issue.includes('phone') || issue.includes('mobile')) return 'phone';
  if (issue.includes('father')) return 'fatherName';
  if (issue.includes('dob') || issue.includes('birth')) return 'dob';
  if (issue.includes('gender')) return 'gender';
  if (issue.includes('nationality')) return 'nationality';
  if (issue.includes('language')) return 'languages';
  if (issue.includes('address') || issue.includes('location')) return 'location';
  if (issue.includes('linkedin')) return 'linkedin';
  if (issue.includes('github')) return 'githubUrl';
  if (issue.includes('name') && !issue.includes('father')) return 'fullName';
  if (issue.includes('certif')) return 'certifications';
  if (issue.includes('project')) return 'projects';
  if (issue.includes('experience') || issue.includes('teaching')) return 'experience';
  if (issue.includes('phd') || issue.includes('degree') || issue.includes('m.tech') || issue.includes('b.tech')) return 'education';
  if (issue.includes('skill')) return 'skill';
  return 'general';
};

    }

import React, { useState } from 'react';
import Screen1Gateway from './components/Screen1Gateway';
import Screen2Form from './components/Screen2Form';
import Screen3Dashboard from './components/Screen3Dashboard';
import './App.css';

export default function App() {
  const [step, setStep] = useState(1);
  const [mode, setMode] = useState('upload');
  const [result, setResult] = useState(null);
  const [profileData, setProfileData] = useState(null);
  const [loading, setLoading] = useState(false);
  const [statusText, setStatusText] = useState('');
  const [error, setError] = useState('');

  const chooseMode = (selectedMode) => {
    setMode(selectedMode);
    setError('');
    setStep(2);
  };

  const onAnalysisDone = (analysisData, initialProfile) => {
    localStorage.removeItem('ats_editor_verified_state');
    localStorage.removeItem('ats_editor_localhost_state');
    setProfileData(initialProfile);
    setResult(analysisData);
    setLoading(false);
    setStep(3);
  };

  const resetAll = () => {
    localStorage.removeItem('ats_editor_verified_state');
    localStorage.removeItem('ats_editor_localhost_state');
    setStep(1);
    setResult(null);
    setProfileData(null);
    setError('');
    setLoading(false);
  };

  return (
    <div className={`app-shell ${step === 3 ? 'dashboard-view' : ''}`}>
      {/* Upper Navigation Bar with Branding on Left and Print Button on Right */}
      <header className="top-nav no-print">
        <div className="nav-container">
          <div className="logo" onClick={resetAll}>
            <span className="badge">ATS</span>
            <span className="logo-title">Resume Editor</span>
          </div>

          {/* Change #2: Print button moved into upper navigation bar */}
          {step === 3 && (
            <button className="btn-primary" onClick={() => window.print()}>
              Print / Save ATS PDF
            </button>
          )}
        </div>
      </header>

      <main className="content-area">
        {error && (
          <div className="error-box no-print">
            <span>{error}</span>
            <button onClick={() => setError('')}>&times;</button>
          </div>
        )}

        {loading && (
          <div className="loading-modal no-print">
            <div className="loader" />
            <p className="loading-title">{statusText || 'Processing...'}</p>
            <p className="loading-sub">Analyzing competencies and tailoring against target Job Description</p>
          </div>
        )}

        {step === 1 && <Screen1Gateway onSelect={chooseMode} />}

        {step === 2 && (
          <Screen2Form
            mode={mode}
            onBack={() => setStep(1)}
            onSuccess={onAnalysisDone}
            setLoading={setLoading}
            setStatusText={setStatusText}
            setError={setError}
          />
        )}

        {step === 3 && (
          <Screen3Dashboard
            data={result}
            fallbackProfile={profileData}
          />
        )}
      </main>
    </div>
  );
}

function checkLinkedInUrl(rawUrl) {
  const tutorialVideoUrl = 'https://www.youtube.com/embed/oga5s3Yngc8';

  if (!rawUrl || typeof rawUrl !== 'string' || !rawUrl.trim()) {
    return {
      issue: 'Missing LinkedIn Profile URL',
      severity: 'MEDIUM',
      suggestion: 'Add your custom shortened LinkedIn URL to your resume header.',
      actionType: 'LINKEDIN_URL_FIX',
      targetField: 'linkedin',
      fieldLabel: 'LinkedIn Profile URL',
      placeholder: 'e.g. linkedin.com/in/yourname',
      videoUrl: tutorialVideoUrl
    };
  }

  const cleaned = rawUrl.replace(/^https?:\/\/(www\.)?/, '').replace(/\/$/, '').trim();
  const hasRandomHash = /-[a-f0-9]{7,}$/i.test(cleaned) || /-\d{6,}$/.test(cleaned);
  const hasQueryParams = cleaned.includes('?');
  const isAbsurdlyLong = cleaned.length > 55;

  if (hasRandomHash || hasQueryParams || isAbsurdlyLong) {
    return {
      issue: 'Uncustomized / Lengthy LinkedIn URL',
      severity: 'LOW',
      suggestion: 'Your LinkedIn URL contains default system numbers. Shorten it, paste your clean handle below, and click Apply.',
      actionType: 'LINKEDIN_URL_FIX',
      targetField: 'linkedin',
      fieldLabel: 'Shortened LinkedIn URL',
      placeholder: 'e.g. linkedin.com/in/yourname',
      videoUrl: tutorialVideoUrl
    };
  }

  return null;
}

function checkBasicInfo(profile = {}) {
  const missingCons = [];
  const pInfo = profile.personalInfo || {};

  const requiredFields = [
    { key: 'fullName', label: 'Full Name', severity: 'HIGH', placeholder: 'Enter Full Name' },
    { key: 'phone', label: 'Phone Number', severity: 'HIGH', placeholder: '+91 XXXXXXXXXX' },
    { key: 'location', label: 'Permanent Address', severity: 'MEDIUM', placeholder: 'City, State' },
    { key: 'githubUrl', label: 'GitHub Profile Link', severity: 'MEDIUM', placeholder: 'github.com/your-username' },
    { key: 'dob', label: 'Date of Birth', severity: 'MEDIUM', placeholder: 'DD/MM/YYYY' },
    { key: 'gender', label: 'Gender', severity: 'LOW', placeholder: 'Male / Female' },
    { key: 'nationality', label: 'Nationality', severity: 'LOW', placeholder: 'Indian' },
    { key: 'fatherName', label: "Father's Name", severity: 'LOW', placeholder: "Father's Name" },
    { key: 'languages', label: 'Languages Known', severity: 'LOW', placeholder: 'English, Hindi, Bengali' }
  ];

  requiredFields.forEach(({ key, label, severity, placeholder }) => {
    const value = pInfo[key];
    if (!value || typeof value !== 'string' || !value.trim()) {
      missingCons.push({
        issue: `Missing ${label}`,
        severity: severity,
        suggestion: `Provide your ${label.toLowerCase()} for the resume header/details.`,
        actionType: 'BASIC_INFO_FIX',
        targetField: key,
        fieldLabel: label,
        placeholder: placeholder
      });
    }
  });

  if (!Array.isArray(profile.certifications) || profile.certifications.length === 0) {
    missingCons.push({
      issue: 'Missing Certifications',
      severity: 'LOW',
      suggestion: 'Adding technical certifications validates your domain expertise to recruiters.',
      actionType: 'BASIC_INFO_FIX',
      targetField: 'certifications',
      fieldLabel: 'Certification (Name - Issuer)',
      placeholder: 'e.g. AWS Certified Developer - Amazon'
    });
  }

  if (!Array.isArray(profile.projects) || profile.projects.length === 0) {
    missingCons.push({
      issue: 'Missing Technical Projects',
      severity: 'HIGH',
      suggestion: 'Add at least one technical project demonstrating your programming skills.',
      actionType: 'BASIC_INFO_FIX',
      targetField: 'projects',
      fieldLabel: 'Project (Title - Stack)',
      placeholder: 'e.g. College Lab App - Python, SQL'
    });
  }

  return missingCons;
}

module.exports = { 
  checkLinkedInUrl,
  checkBasicInfo 
};

const express = require('express');
const router = express.Router();
const multer = require('multer');
const pdfParse = require('pdf-parse');
const axios = require('axios');

const Profile = require('../models/Profile');
const JobAnalysis = require('../models/JobAnalysis');
const { checkLinkedInUrl, checkBasicInfo } = require('../util');

const upload = multer({ 
    storage: multer.memoryStorage(),
    limits: { fileSize: 10 * 1024 * 1024 }
});

const AI_SERVICE_URL = (process.env.AI_SERVICE_URL || 'http://localhost:8000').trim().replace(/\/$/, '');

function extractCleanJobTitle(jdText = '', aiTitle = '') {
    if (aiTitle && !/^(about the job|job description|target role|professional|overview|ugc norms)$/i.test(aiTitle.trim())) {
        return aiTitle.trim();
    }

    const lines = jdText.split('\n').map(l => l.trim()).filter(Boolean);
    
    for (const line of lines) {
        const match = line.match(/^(?:job title|role|position|designation|hiring for)\s*[:\-]\s*(.+)$/i);
        if (match && match[1]) {
            const candidate = match[1].trim();
            if (!/about the job|ugc/i.test(candidate)) return candidate.slice(0, 45);
        }
    }

    const titleRegex = /\b(?:frontend|backend|full[\s-]?stack|software|web|react|node|python|java|systems?|data|machine learning|ai|devops|cloud|qa|test|assistant professor|lecturer)\s+(?:developer|engineer|architect|lead|specialist|consultant|intern|educator|professor)\b/i;
    for (const line of lines) {
        if (/about the job|ugc/i.test(line)) continue;
        const match = line.match(titleRegex);
        if (match) return match[0].split(' ').map(w => w.charAt(0).toUpperCase() + w.slice(1).toLowerCase()).join(' ');
    }

    return 'Full Stack - Web Developer';
}

router.post('/profiles/manual', async (req, res) => {
    try {
        const newProfile = new Profile(req.body);
        const savedProfile = await newProfile.save();
        res.status(201).json({ success: true, profileId: savedProfile._id, profileData: savedProfile });
    } catch (error) {
        res.status(422).json({ success: false, message: error.message || 'Validation error' });
    }
});

router.post('/profiles/upload-pdf', upload.single('file'), async (req, res) => {
    try {
        if (!req.file) return res.status(400).json({ success: false, message: 'PDF file required.' });

        const parsedPdf = await pdfParse(req.file.buffer);
        const rawText = parsedPdf.text;
        if (!rawText || !rawText.trim()) return res.status(400).json({ success: false, message: 'Could not extract text.' });

        const aiResponse = await axios.post(`${AI_SERVICE_URL}/ai/parse-linkedin`, { rawText });
        const newProfile = new Profile(aiResponse.data);
        const savedProfile = await newProfile.save();

        res.status(201).json({ success: true, profileId: savedProfile._id, profileData: savedProfile });
    } catch (error) {
        res.status(500).json({ success: false, message: 'Error processing PDF.' });
    }
});

router.post(['/profiles/:id/analyze', '/profiles/:id/analyse'], async (req, res) => {
    try {
        const { id } = req.params;
        const jobDescription = req.body.jobDescription || req.body.jobdescription;
        if (!jobDescription || !jobDescription.trim()) {
            return res.status(400).json({ success: false, message: 'Job Description is required.' });
        }

        const profile = await Profile.findById(id);
        if (!profile) return res.status(404).json({ success: false, message: 'Profile not found.' });

        const aiResponse = await axios.post(`${AI_SERVICE_URL}/ai/analyze-and-tailor`, {
            profileData: profile,
            jobDescription: jobDescription
        });

        const aiData = aiResponse.data;
        const pInfo = profile.personalInfo || {};

        // 1. Run deterministic checks
        const linkedInCon = checkLinkedInUrl(pInfo.linkedinUrl);
        const basicInfoCons = checkBasicInfo(profile.toObject());

        // 2. Filter out false experience gaps, generic personal details, & UGC salary noise
        const rawAiCons = aiData.audit?.cons || [];
        const filteredAiCons = rawAiCons.filter(c => {
            const text = `${c.issue || ''} ${c.suggestion || ''}`.toLowerCase();
            if (text.includes('personal detail') || text.includes('contact information')) return false;
            if (text.includes('salary') || text.includes('ugc')) return false;
            if (/\b(\d+\+?\s*(?:to\s*\d+\s*)?years?|\d+\s*yrs?)\b/i.test(text) && text.includes('experience')) return false;
            return true;
        });

        // 3. Filter durations from missing keywords
        const filteredKeywords = (aiData.audit?.missingKeywords || []).filter(kw => {
            return !/\b(\d+\+?\s*years?|\d+\s*yrs?|experience|salary|ugc)\b/i.test(kw);
        });

        const finalCons = [...basicInfoCons];
        if (linkedInCon) finalCons.unshift(linkedInCon);
        finalCons.push(...filteredAiCons);

        // 4. Extract clean job title (Never "About the job")
        const targetJobTitle = extractCleanJobTitle(jobDescription, aiData.targetJobTitle);

        // 5. Sanitize summary from salary / UGC norms
        if (aiData.tailoredContent?.tailoredSummary) {
            aiData.tailoredContent.tailoredSummary = aiData.tailoredContent.tailoredSummary
                .replace(/salary\s*(as\s*per\s*)?ugc\s*norms/gi, '')
                .replace(/\s+/g, ' ')
                .trim();
        }

        const newAnalysis = new JobAnalysis({
            profileId: profile._id,
            targetJob: {
                jobTitle: targetJobTitle,
                rawJobDescription: jobDescription
            },
            scoring: aiData.scoring,
            audit: {
                pros: aiData.audit?.pros || [],
                cons: finalCons,
                missingKeywords: filteredKeywords
            },
            tailoredResume: aiData.tailoredResume || aiData.tailoredContent
        });
        const savedAnalysis = await newAnalysis.save();

        res.status(200).json({
            success: true,
            analysisId: savedAnalysis._id,
            profileId: profile._id,
            targetJob: savedAnalysis.targetJob,
            scoring: savedAnalysis.scoring,
            audit: savedAnalysis.audit,
            tailoredResume: savedAnalysis.tailoredResume
        });
    } catch (error) {
        console.error('Analysis error:', error);
        res.status(500).json({ success: false, message: 'Analysis failed.' });
    }
    
    // Clean dynamic title from AI (Fallback to clean line extraction only if AI is offline)
    let finalJobTitle = (aiData.targetJobTitle || '').trim();

    if (!finalJobTitle || /^(about the job|overview|job description|ugc norms)$/i.test(finalJobTitle)) {
        // Dynamic fallback: scan lines for "Role: ...", "Position: ..." or first clean headline
        const lines = jobDescription.split('\n').map(l => l.trim()).filter(Boolean);
        for (const line of lines) {
            const match = line.match(/^(?:job title|role|position|designation|hiring for)\s*[:\-]\s*(.+)$/i);
            if (match && match[1] && !/about the job/i.test(match[1])) {
                finalJobTitle = match[1].trim().slice(0, 45);
                break;
            }
        }
        if (!finalJobTitle) {
            for (const line of lines) {
                if (!/^(about the job|job description|overview|requirements|responsibilities|details|we are hiring):?/i.test(line) && line.length > 2 && line.length < 45) {
                    finalJobTitle = line;
                    break;
                }
            }
        }
    }

    if (!finalJobTitle) finalJobTitle = 'Professional';

    const newAnalysis = new JobAnalysis({
        profileId: profile._id,
        targetJob: {
            jobTitle: finalJobTitle, // DYNAMIC TITLE
            rawJobDescription: jobDescription
        },
        scoring: aiData.scoring,
        audit: {
            pros: aiData.audit?.pros || [],
            cons: finalCons,
            missingKeywords: filteredKeywords
        },
        tailoredResume: aiData.tailoredResume || aiData.tailoredContent
});
const savedAnalysis = await newAnalysis.save();

res.status(200).json({
    success: true,
    analysisId: savedAnalysis._id,
    profileId: profile._id,
    targetJob: savedAnalysis.targetJob, // Sent to frontend
    scoring: savedAnalysis.scoring,
    audit: savedAnalysis.audit,
    tailoredResume: savedAnalysis.tailoredResume
});
});

router.patch('/profiles/:id/basic-info', async (req, res) => {
    try {
        const { field, value } = req.body;
        if (!field) return res.status(400).json({ success: false, message: 'Field name required.' });

        const cleanVal = typeof value === 'string' ? value.replace(/^https?:\/\/(www\.)?/, '').replace(/\/$/, '').trim() : value;
        const updateKey = `personalInfo.${field}`;

        const updated = await Profile.findByIdAndUpdate(
            req.params.id,
            { $set: { [updateKey]: cleanVal } },
            { new: true }
        );
        res.status(200).json({ success: true, profile: updated });
    } catch (err) {
        res.status(500).json({ success: false, message: 'Update failed.' });
    }
});
